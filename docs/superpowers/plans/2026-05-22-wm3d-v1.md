> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a 3D-native world model with scene generation + scene prediction + VLA action output, on frozen VGGT-1B + Qwen3-VL-Embedding-2B, trained and evaluated on LIBERO.

**Architecture:** DualStreamDynamics — parallel state + action transformers with periodic bidirectional cross-attention. State stream predicts future VGGT tokens (k=8 ahead from T=16 input). Action stream is an IDM emitting continuous latent z_a, decoded by a small projection head into 7-DoF actions. Geometry decoded via small auxiliary head from pooled tokens.

**Tech Stack:** PyTorch 2.x, transformers (Qwen3-VL), VGGT (facebook/VGGT-1B via existing `world_model/src/vggt_wrapper.py`), lerobot dataset format, robosuite/LIBERO for closed-loop eval, bf16, torch.distributed DDP, TensorBoard.

**Spec:** `docs/superpowers/specs/2026-05-22-wm3d-v1-design.md`. All design decisions are frozen there.

```python
from __future__ import annotations
import json
from dataclasses import dataclass, asdict, field
from pathlib import Path

@dataclass
class ClipRecord:
    clip_id: str
    dataset: str
    split: str
    embodiment: str
    n_frames: int
    fps: int
    video_paths: dict[str, str]
    data_parquet: str
    task_text: str
    action_dim: int
    action_kind: str

def write_manifest(path: str | Path, records: list[ClipRecord]) -> None:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open("w") as f:
        for r in records:
            f.write(json.dumps(asdict(r)) + "\n")

def read_manifest(path: str | Path) -> list[ClipRecord]:
    out: list[ClipRecord] = []
    with Path(path).open() as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            out.append(ClipRecord(**json.loads(line)))
    return out
```

```python
from __future__ import annotations
import json
from pathlib import Path
import pyarrow.parquet as pq
from .manifest import ClipRecord

def ingest_libero_split(
    root: Path, split_name: str, max_episodes: int | None = None
) -> list[ClipRecord]:
    info = json.loads((root / "meta" / "info.json").read_text())
    fps = int(info["fps"])
    # task strings live in meta/tasks.jsonl; map task_index → task
    task_map: dict[int, str] = {}
    tasks_path = root / "meta" / "tasks.jsonl"
    if tasks_path.exists():
        for line in tasks_path.read_text().splitlines():
            j = json.loads(line)
            task_map[int(j["task_index"])] = j.get("task", "")
    # episodes.jsonl maps episode_index → task_index, n_frames
    eps_path = root / "meta" / "episodes.jsonl"
    records: list[ClipRecord] = []
    with eps_path.open() as f:
        for line in f:
            ep = json.loads(line)
            ep_idx = int(ep["episode_index"])
            chunk_idx = ep_idx // int(info["chunks_size"])
            n_frames = int(ep["length"])
            data_parquet = root / info["data_path"].format(
                episode_chunk=chunk_idx, episode_index=ep_idx
            )
            video_main = root / info["video_path"].format(
                episode_chunk=chunk_idx, video_key="observation.images.image",
                episode_index=ep_idx,
            )
            video_wrist = root / info["video_path"].format(
                episode_chunk=chunk_idx, video_key="observation.images.wrist_image",
                episode_index=ep_idx,
            )
            tasks = ep.get("tasks", [])
            if tasks and isinstance(tasks[0], int):
                task_text = task_map.get(tasks[0], "")
            else:
                task_text = tasks[0] if tasks else ""
            rec = ClipRecord(
                clip_id=f"{split_name}/ep_{ep_idx:06d}",
                dataset="libero",
                split=split_name,
                embodiment="franka",
                n_frames=n_frames,
                fps=fps,
                video_paths={"main": str(video_main), "wrist": str(video_wrist)},
                data_parquet=str(data_parquet),
                task_text=task_text,
                action_dim=7,
                action_kind="delta_pose+gripper",
            )
            records.append(rec)
            if max_episodes is not None and len(records) >= max_episodes:
                break
    return records
```

```python
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
from typing import Callable, Any
import numpy as np
import torch
import pyarrow.parquet as pq
from torch.utils.data import Dataset
from .manifest import ClipRecord

@dataclass
class WindowConfig:
    T: int = 16
    k: int = 8
    stride: int = 4
    cache_root: Path = Path("/home/user01/Minko/datasets/cache/wm3d")
    pooled_subdir: str = "vggt_pooled"
    geom_subdir: str = "vggt_geom"
    qwen_subdir: str = "qwen_taskemb"

def _default_resolver(rec: ClipRecord, kind: str, cfg: WindowConfig) -> Any:
    cid = rec.clip_id.replace("/", "__")
    if kind == "pooled":
        return np.load(cfg.cache_root / cfg.pooled_subdir / f"{cid}.npy", mmap_mode="r")
    if kind == "geom":
        return np.load(cfg.cache_root / cfg.geom_subdir / f"{cid}.npz")
    if kind == "qwen":
        return np.load(cfg.cache_root / cfg.qwen_subdir / f"{cid}.npy", mmap_mode="r")
    if kind == "action":
        return Path(rec.data_parquet)
    raise KeyError(kind)

class WindowDataset(Dataset):
    def __init__(self, records: list[ClipRecord], cfg: WindowConfig,
                 _resolver: Callable[[ClipRecord, str], Any] | None = None):
        self.cfg = cfg
        self.records = records
        self._resolver = _resolver or (lambda r, kind: _default_resolver(r, kind, cfg))
        self.index: list[tuple[int, int]] = []  # (record_idx, start_frame)
        win = cfg.T + cfg.k
        for i, r in enumerate(records):
            if r.n_frames < win:
                continue
            for start in range(0, r.n_frames - win + 1, cfg.stride):
                self.index.append((i, start))

    def __len__(self) -> int:
        return len(self.index)

    def __getitem__(self, i: int) -> dict[str, torch.Tensor]:
        ri, start = self.index[i]
        rec = self.records[ri]
        T, k = self.cfg.T, self.cfg.k
        pooled = self._resolver(rec, "pooled")[start:start + T + k]   # [T+k, P, D]
        geom = self._resolver(rec, "geom")
        qwen = self._resolver(rec, "qwen")
        parquet_path = self._resolver(rec, "action")
        # Lazy parquet load for the slice — read all actions then slice
        action_table = pq.read_table(parquet_path, columns=["action"])
        actions = np.stack(action_table.column("action").to_pylist()).astype(np.float32)
        a_slice = actions[start + T : start + T + k]  # k actions after the input window
        sample = {
            "s_in":     torch.from_numpy(np.ascontiguousarray(pooled[:T])).float(),
            "s_tgt":    torch.from_numpy(np.ascontiguousarray(pooled[T:])).float(),
            "depth_tgt": torch.from_numpy(np.ascontiguousarray(geom["depth"][start + T : start + T + k])).float(),
            "point_tgt": torch.from_numpy(np.ascontiguousarray(geom["point"][start + T : start + T + k])).float(),
            "pose_tgt":  torch.from_numpy(np.ascontiguousarray(geom["pose"][start + T : start + T + k])).float(),
            "action_tgt": torch.from_numpy(a_slice).float(),
            "c":         torch.from_numpy(np.ascontiguousarray(np.asarray(qwen))).float(),
            "clip_id":   rec.clip_id,
            "start":     start,
        }
        return sample
```

```python
from __future__ import annotations
import sys
from pathlib import Path
import torch
import torch.nn as nn
import torch.nn.functional as F

# Bring world_model/src on path to reuse the existing VGGT wrapper.
WM_SRC = Path("/home/user01/Minko/newwm/world_model/src")
if str(WM_SRC) not in sys.path:
    sys.path.insert(0, str(WM_SRC))
from vggt_wrapper import VGGTWrapper, VGGTConfig  # noqa: E402

class VGGTEncoder(nn.Module):
    """Wraps VGGT-1B; exports pooled 8x8 tokens + per-frame geometry."""

    def __init__(self, device: str = "cuda", token_grid: int = 8):
        super().__init__()
        self.token_grid = token_grid
        self.wrapper = VGGTWrapper(VGGTConfig(device=device))

    @torch.no_grad()
    def forward(self, images: torch.Tensor) -> dict:
        # images: [B, T, 3, H, W] in [0, 1]
        B, T = images.shape[:2]
        flat = images.flatten(0, 1)
        out = self.wrapper.full_inference(flat)
        aggregated_tokens = out["aggregated_tokens"]
        # Take last-layer aggregator output, reshape patch grid, average-pool to 8×8.
        # VGGT aggregator output shape: each layer is [B, n_tokens, D]; we use the last layer.
        last = aggregated_tokens[-1]
        # n_tokens = special + img_patches. We slice the image patch tokens via ps_idx.
        ps_idx = out["ps_idx"]
        img_tokens = last[:, ps_idx[0]]  # [B*T, G*G_full, D]; ps_idx is 1-D over image patches
        # Reshape to spatial grid then pool
        BT = img_tokens.shape[0]
        D = img_tokens.shape[-1]
        n_patch = img_tokens.shape[1]
        side = int(n_patch ** 0.5)
        img_tokens = img_tokens.view(BT, side, side, D).permute(0, 3, 1, 2)
        pooled = F.adaptive_avg_pool2d(img_tokens, self.token_grid)  # [BT, D, 8, 8]
        pooled = pooled.permute(0, 2, 3, 1).reshape(B, T, self.token_grid * self.token_grid, D)
        # Geometry — downsample depth/point to 64×64 for cache compactness
        depth = out["depth"].view(B, T, *out["depth"].shape[-2:])
        depth_64 = F.adaptive_avg_pool2d(depth, 64)
        point = out["point_map"].view(B, T, *out["point_map"].shape[-3:])
        point = point.permute(0, 1, 4, 2, 3)  # [B,T,3,H,W]
        point_64 = F.adaptive_avg_pool2d(point.flatten(0, 1), 64).view(B, T, 3, 64, 64).permute(0, 1, 3, 4, 2)
        # Pose: stack extrinsic (rotation 6-d via 2 cols) + translation (3) = 9
        extri = out["camera_extrinsic"].view(B, T, 3, 4)
        rot6 = extri[..., :3, :2].reshape(B, T, 6)
        trans = extri[..., :, 3]
        pose = torch.cat([rot6, trans], dim=-1)  # [B, T, 9]
        return {"pooled": pooled.to(torch.float32),
                "depth":  depth_64.to(torch.float32),
                "point":  point_64.to(torch.float32),
                "pose":   pose.to(torch.float32)}
```

```python
from pathlib import Path
import torch
import torch.nn.functional as F
from PIL import Image
from transformers import AutoProcessor, AutoModel

DEFAULT_PATH = Path("/home/user01/Minko/models/Qwen3-VL-Embedding-2B")

class QwenVLEmbed:
    def __init__(self, model_path: Path = DEFAULT_PATH, device: str = "cuda",
                 dim: int = 2048):
        self.device = device
        self.dim = dim
        self.processor = AutoProcessor.from_pretrained(model_path, trust_remote_code=True)
        self.model = AutoModel.from_pretrained(
            model_path, trust_remote_code=True, torch_dtype=torch.bfloat16
        ).to(device).eval()

    @torch.no_grad()
    def embed(self, text: str, image: Image.Image | None = None) -> torch.Tensor:
        messages = [{"role": "user", "content": []}]
        if image is not None:
            messages[0]["content"].append({"type": "image", "image": image})
        messages[0]["content"].append({"type": "text", "text": text})
        chat = self.processor.apply_chat_template(messages, tokenize=False,
                                                   add_generation_prompt=False)
        inputs = self.processor(text=[chat],
                                images=[image] if image is not None else None,
                                return_tensors="pt", padding=True).to(self.device)
        out = self.model(**inputs, output_hidden_states=True)
        hidden = out.hidden_states[-1]  # [B, L, D]
        mask = inputs.get("attention_mask", torch.ones_like(hidden[..., 0]))
        # Last-token pool (Qwen embedding convention)
        idx = mask.sum(dim=1) - 1
        pooled = hidden[torch.arange(hidden.size(0)), idx]
        pooled = F.normalize(pooled.float(), dim=-1)
        return pooled[0, : self.dim].contiguous().cpu()
```

```python
import torch
import torch.nn as nn

class BidirectionalCrossAttn(nn.Module):
    def __init__(self, d_state: int = 1024, d_action: int = 1024, n_heads: int = 16,
                 dropout: float = 0.0):
        super().__init__()
        self.s_norm = nn.LayerNorm(d_state)
        self.a_norm = nn.LayerNorm(d_action)
        self.s_to_a = nn.MultiheadAttention(d_state, n_heads, dropout=dropout,
                                            kdim=d_action, vdim=d_action,
                                            batch_first=True)
        self.a_to_s = nn.MultiheadAttention(d_action, n_heads, dropout=dropout,
                                            kdim=d_state, vdim=d_state,
                                            batch_first=True)
        self.s_proj = nn.Linear(d_state, d_state)
        self.a_proj = nn.Linear(d_action, d_action)

    def forward(self, s: torch.Tensor, a: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        sn, an = self.s_norm(s), self.a_norm(a)
        s_upd, _ = self.s_to_a(sn, an, an, need_weights=False)
        a_upd, _ = self.a_to_s(an, sn, sn, need_weights=False)
        s = s + self.s_proj(s_upd)
        a = a + self.a_proj(a_upd)
        return s, a
```

```python
from __future__ import annotations
from dataclasses import dataclass
import torch
import torch.nn as nn

@dataclass
class StateConfig:
    T: int = 16
    P: int = 64
    D: int = 2048
    hidden: int = 1024
    n_layers: int = 12
    n_heads: int = 16
    k: int = 8
    mlp_mult: int = 4
    dropout: float = 0.0
    cond_dim: int = 2048

class StateStream(nn.Module):
    def __init__(self, cfg: StateConfig):
        super().__init__()
        self.cfg = cfg
        self.in_proj = nn.Linear(cfg.D, cfg.hidden)
        self.cond_proj = nn.Linear(cfg.cond_dim, cfg.hidden)
        self.frame_pos = nn.Parameter(torch.zeros(1, cfg.T, 1, cfg.hidden))
        self.patch_pos = nn.Parameter(torch.zeros(1, 1, cfg.P, cfg.hidden))
        nn.init.normal_(self.frame_pos, std=0.02)
        nn.init.normal_(self.patch_pos, std=0.02)
        layer = nn.TransformerEncoderLayer(
            d_model=cfg.hidden, nhead=cfg.n_heads,
            dim_feedforward=cfg.hidden * cfg.mlp_mult,
            dropout=cfg.dropout, activation="gelu",
            batch_first=True, norm_first=True,
        )
        self.layers = nn.ModuleList([
            nn.TransformerEncoderLayer(
                d_model=cfg.hidden, nhead=cfg.n_heads,
                dim_feedforward=cfg.hidden * cfg.mlp_mult,
                dropout=cfg.dropout, activation="gelu",
                batch_first=True, norm_first=True,
            ) for _ in range(cfg.n_layers)
        ])
        self.norm = nn.LayerNorm(cfg.hidden)
        # decoder: k learnable queries cross-attend to encoder output
        self.dec_q = nn.Parameter(torch.zeros(1, cfg.k, cfg.hidden))
        nn.init.normal_(self.dec_q, std=0.02)
        dec_layer = nn.TransformerDecoderLayer(
            d_model=cfg.hidden, nhead=cfg.n_heads,
            dim_feedforward=cfg.hidden * cfg.mlp_mult,
            dropout=cfg.dropout, activation="gelu",
            batch_first=True, norm_first=True,
        )
        self.decoder = nn.TransformerDecoder(dec_layer, num_layers=2)
        self.out_proj = nn.Linear(cfg.hidden, cfg.P * cfg.D)

    def encode(self, s: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
        # s: [B, T, P, D]; c: [B, cond_dim]
        B = s.size(0)
        x = self.in_proj(s)
        x = x + self.frame_pos + self.patch_pos
        x = x.view(B, self.cfg.T * self.cfg.P, self.cfg.hidden)
        cond_tok = self.cond_proj(c).unsqueeze(1)
        x = torch.cat([cond_tok, x], dim=1)
        return x

    def apply_blocks_with_xattn(self, x: torch.Tensor, xattn_callbacks=None) -> torch.Tensor:
        for i, layer in enumerate(self.layers):
            x = layer(x)
            if xattn_callbacks and i in xattn_callbacks:
                x = xattn_callbacks[i](x)
        x = self.norm(x)
        return x

    def decode(self, h: torch.Tensor) -> torch.Tensor:
        B = h.size(0)
        q = self.dec_q.expand(B, -1, -1)
        out = self.decoder(q, h)             # [B, k, hidden]
        pred = self.out_proj(out)             # [B, k, P*D]
        return pred.view(B, self.cfg.k, self.cfg.P, self.cfg.D)

    def forward(self, s: torch.Tensor, c: torch.Tensor):
        x = self.encode(s, c)
        x = self.apply_blocks_with_xattn(x)
        pred = self.decode(x)
        return x, pred
```

```python
from __future__ import annotations
from dataclasses import dataclass, field
import torch
import torch.nn as nn
from .state_stream import StateStream, StateConfig
from .action_stream import ActionStream, ActionConfig
from .cross_attn import BidirectionalCrossAttn

@dataclass
class DualConfig:
    state: StateConfig = field(default_factory=StateConfig)
    action: ActionConfig = field(default_factory=ActionConfig)
    xattn_layers_state: tuple[int, ...] = (3, 6, 9, 11)
    xattn_n_heads: int = 16

class DualStreamDynamics(nn.Module):
    def __init__(self, cfg: DualConfig):
        super().__init__()
        self.cfg = cfg
        self.state = StateStream(cfg.state)
        self.action = ActionStream(cfg.action)
        # Match action stream layer count to xattn schedule
        n_xattn = len(cfg.xattn_layers_state)
        action_indices = sorted(set(int(round((cfg.action.n_layers - 1) * i /
                                              max(1, cfg.state.n_layers - 1)))
                                    for i in cfg.xattn_layers_state))
        self.action_xattn_indices = action_indices
        # action_proj_up brings action hidden to state hidden for cross-attn
        self.action_to_state_dim = nn.Linear(cfg.action.hidden, cfg.state.hidden) \
            if cfg.action.hidden != cfg.state.hidden else nn.Identity()
        self.state_to_action_dim = nn.Linear(cfg.state.hidden, cfg.action.hidden) \
            if cfg.action.hidden != cfg.state.hidden else nn.Identity()
        self.xattn_blocks = nn.ModuleList([
            BidirectionalCrossAttn(d_state=cfg.state.hidden, d_action=cfg.state.hidden,
                                   n_heads=cfg.xattn_n_heads)
            for _ in range(n_xattn)
        ])

    def forward(self, s: torch.Tensor, c: torch.Tensor) -> dict[str, torch.Tensor]:
        # Encode initial hidden states for both streams.
        h_s = self.state.encode(s, c)
        h_a = self.action.encode(s, c)
        # Interleave layers + cross-attn at matched depths.
        xattn_iter = iter(zip(self.cfg.xattn_layers_state,
                              self.action_xattn_indices, self.xattn_blocks))
        next_x = next(xattn_iter, None)
        # Build per-stream layer iteration with cross-attn hooks.
        s_i = a_i = 0
        # Pre-compute layer schedule for action stream interleave.
        a_layers_per_state = max(1, self.cfg.action.n_layers // self.cfg.state.n_layers)
        for s_i, layer_s in enumerate(self.state.layers):
            h_s = layer_s(h_s)
            # Step the action stream to keep depths comparable.
            for _ in range(a_layers_per_state):
                if a_i < len(self.action.layers):
                    h_a = self.action.layers[a_i](h_a)
                    a_i += 1
            if next_x is not None and s_i == next_x[0]:
                xs_i, xa_i, blk = next_x
                h_a_up = self.action_to_state_dim(h_a)
                h_s_new, h_a_up_new = blk(h_s, h_a_up)
                h_s = h_s_new
                h_a = self.state_to_action_dim(h_a_up_new)
                next_x = next(xattn_iter, None)
        # Flush remaining action layers if any
        while a_i < len(self.action.layers):
            h_a = self.action.layers[a_i](h_a)
            a_i += 1
        h_s = self.state.norm(h_s)
        h_a = self.action.norm(h_a)
        pred = self.state.decode(h_s)
        z_a = self.action.decode(h_a)
        return {"pred_tokens": pred, "z_a": z_a, "h_s": h_s, "h_a": h_a}
```

```python
from __future__ import annotations
import torch
import torch.nn as nn

class ActionProjHead(nn.Module):
    def __init__(self, z_dim: int = 128, hidden: int = 512, n_layers: int = 4,
                 action_dim: int = 7):
        super().__init__()
        layers: list[nn.Module] = [nn.Linear(z_dim, hidden), nn.GELU()]
        for _ in range(n_layers - 2):
            layers.extend([nn.Linear(hidden, hidden), nn.GELU()])
        self.trunk = nn.Sequential(*layers)
        self.pose_head = nn.Linear(hidden, action_dim - 1)
        self.grip_head = nn.Linear(hidden, 1)

    def forward(self, z: torch.Tensor) -> dict[str, torch.Tensor]:
        h = self.trunk(z)
        pose = torch.tanh(self.pose_head(h))
        grip_logit = self.grip_head(h).squeeze(-1)
        return {"pose": pose, "gripper_logit": grip_logit}
```

```python
from __future__ import annotations
import torch
import torch.nn as nn
import torch.nn.functional as F

class PooledTokenGeomDecoder(nn.Module):
    """[B, T, 64, 2048] -> depth [B,T,64,64], point [B,T,64,64,3], pose [B,T,9]."""
    def __init__(self, token_dim: int = 2048, token_grid: int = 8, hidden: int = 256):
        super().__init__()
        self.token_grid = token_grid
        H = hidden
        self.stem = nn.Conv2d(token_dim, H, 1)
        self.up1 = nn.Sequential(
            nn.ConvTranspose2d(H, H, 4, 2, 1), nn.GELU(),
            nn.Conv2d(H, H, 3, padding=1), nn.GELU(),
        )
        self.up2 = nn.Sequential(
            nn.ConvTranspose2d(H, H // 2, 4, 2, 1), nn.GELU(),
            nn.Conv2d(H // 2, H // 2, 3, padding=1), nn.GELU(),
        )
        self.up3 = nn.Sequential(
            nn.ConvTranspose2d(H // 2, H // 4, 4, 2, 1), nn.GELU(),
            nn.Conv2d(H // 4, H // 4, 3, padding=1), nn.GELU(),
        )
        self.depth_head = nn.Conv2d(H // 4, 1, 1)
        self.point_head = nn.Conv2d(H // 4, 3, 1)
        # Pose head: global avg pool of token stem → MLP
        self.pose_head = nn.Sequential(
            nn.AdaptiveAvgPool2d(1), nn.Flatten(),
            nn.Linear(H, H), nn.GELU(),
            nn.Linear(H, 9),
        )

    def forward(self, s: torch.Tensor) -> dict[str, torch.Tensor]:
        # s: [B, T, P, D]; reshape to per-frame [B*T, D, G, G]
        B, T, P, D = s.shape
        G = self.token_grid
        x = s.reshape(B * T, G * G, D).transpose(1, 2).reshape(B * T, D, G, G)
        x = self.stem(x)
        pose = self.pose_head(x).view(B, T, 9)
        x = self.up1(x); x = self.up2(x); x = self.up3(x)
        depth = F.softplus(self.depth_head(x)).squeeze(1).view(B, T, 64, 64)
        point = self.point_head(x).permute(0, 2, 3, 1).contiguous().view(B, T, 64, 64, 3)
        return {"depth": depth, "point": point, "pose": pose}
```

```python
from __future__ import annotations
from dataclasses import dataclass
import torch
import torch.nn as nn
import torch.nn.functional as F

@dataclass
class LossWeights:
    cos: float = 0.1
    grip: float = 0.5
    geom_depth: float = 0.2
    geom_point: float = 0.2
    geom_pose: float = 0.1
    action: float = 1.0
    idm_reg: float = 0.01

def _normalize_depth(d: torch.Tensor) -> torch.Tensor:
    med = d.flatten(-2).median(dim=-1).values.clamp_min(1e-6)
    return d / med[..., None, None]

def compute_losses(out: dict, tgt: dict, w: LossWeights) -> dict[str, torch.Tensor]:
    pred_s = out["pred_tokens"]
    s_tgt = tgt["s_tgt"].to(pred_s.dtype)
    L_state_mse = F.mse_loss(pred_s, s_tgt)
    cos = F.cosine_similarity(pred_s.flatten(-2), s_tgt.flatten(-2), dim=-1).mean()
    L_state = L_state_mse + w.cos * (1.0 - cos)
    depth_pred = out["depth"]
    depth_tgt = tgt["depth_tgt"]
    L_depth = F.l1_loss(_normalize_depth(depth_pred), _normalize_depth(depth_tgt))
    L_point = F.l1_loss(out["point"], tgt["point_tgt"])
    L_pose_g = F.mse_loss(out["pose_geom"], tgt["pose_tgt"])
    L_geom = (w.geom_depth * L_depth + w.geom_point * L_point + w.geom_pose * L_pose_g)
    a_tgt = tgt["action_tgt"]
    L_pose_a = F.mse_loss(out["pose"], a_tgt[..., :6])
    L_grip = F.binary_cross_entropy_with_logits(out["gripper_logit"], a_tgt[..., 6])
    L_action = L_pose_a + w.grip * L_grip
    L_idm = (out["z_a"] ** 2).mean()
    L_total = L_state + L_geom + w.action * L_action + w.idm_reg * L_idm
    return {
        "L_total": L_total,
        "L_state": L_state.detach(),
        "L_state_mse": L_state_mse.detach(),
        "L_cos": (1 - cos).detach(),
        "L_geom": L_geom.detach(),
        "L_depth": L_depth.detach(),
        "L_point": L_point.detach(),
        "L_pose_geom": L_pose_g.detach(),
        "L_action": L_action.detach(),
        "L_pose_action": L_pose_a.detach(),
        "L_grip": L_grip.detach(),
        "L_idm": L_idm.detach(),
    }
```

```python
"""Train wm3d v1. Single-GPU or torchrun-DDP."""
from __future__ import annotations
import argparse, os, math, json, time
from pathlib import Path
import yaml
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, Subset
from torch.utils.tensorboard import SummaryWriter
import torch.distributed as dist
from torch.nn.parallel import DistributedDataParallel as DDP

from wm3d.data.manifest import read_manifest
from wm3d.data.window_dataset import WindowDataset, WindowConfig
from wm3d.models.wm3d_model import WM3DModel, WM3DConfig
from wm3d.models.dual_stream import DualConfig
from wm3d.models.state_stream import StateConfig
from wm3d.models.action_stream import ActionConfig
from wm3d.losses import compute_losses, LossWeights

def setup_ddp() -> tuple[int, int, int]:
    if "RANK" in os.environ:
        dist.init_process_group("nccl")
        rank = dist.get_rank()
        world = dist.get_world_size()
        local = int(os.environ.get("LOCAL_RANK", rank))
        torch.cuda.set_device(local)
        return rank, world, local
    return 0, 1, 0

def build_model(cfg: dict) -> WM3DModel:
    sc = StateConfig(**cfg["model"]["state"])
    ac = ActionConfig(**cfg["model"]["action"])
    dc = DualConfig(state=sc, action=ac,
                    xattn_layers_state=tuple(cfg["model"]["xattn_layers_state"]))
    mc = WM3DConfig(dual=dc,
                    action_proj_hidden=cfg["model"]["action_proj_hidden"],
                    action_proj_layers=cfg["model"]["action_proj_layers"],
                    geom_hidden=cfg["model"]["geom_hidden"])
    return WM3DModel(mc)

def build_datasets(cfg: dict, overfit_ids: list[str] | None = None):
    records = read_manifest(cfg["data"]["manifest"])
    if overfit_ids:
        records = [r for r in records if r.clip_id in overfit_ids]
    wcfg = WindowConfig(T=cfg["data"]["T"], k=cfg["data"]["k"],
                        stride=cfg["data"]["stride"],
                        cache_root=Path(cfg["data"]["cache_root"]))
    ds = WindowDataset(records, wcfg)
    n = len(ds)
    g = torch.Generator().manual_seed(cfg["data"]["seed"])
    perm = torch.randperm(n, generator=g).tolist()
    n_val = max(1, int(n * cfg["data"]["val_frac"]))
    val = Subset(ds, perm[:n_val])
    tr = Subset(ds, perm[n_val:])
    return tr, val

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--cfg", type=Path, required=True)
    ap.add_argument("--overfit", action="store_true",
                    help="ignore train data, train on overfit_clip_ids only")
    args = ap.parse_args()
    cfg = yaml.safe_load(args.cfg.read_text())
    rank, world, local = setup_ddp()
    device = torch.device(f"cuda:{local}")

    overfit_ids = cfg.get("overfit_clip_ids") if args.overfit else None
    train_ds, val_ds = build_datasets(cfg, overfit_ids=overfit_ids)
    bs = cfg["train"]["batch_size_per_gpu"]
    train_loader = DataLoader(train_ds, batch_size=bs, shuffle=True,
                              num_workers=cfg["train"]["num_workers"],
                              pin_memory=True, drop_last=True)
    val_loader = DataLoader(val_ds, batch_size=bs, shuffle=False,
                            num_workers=cfg["train"]["num_workers"],
                            pin_memory=True)
    model = build_model(cfg).to(device)
    if world > 1:
        model = DDP(model, device_ids=[local], find_unused_parameters=False)
    opt = torch.optim.AdamW(model.parameters(), lr=cfg["train"]["lr"],
                            weight_decay=cfg["train"]["weight_decay"],
                            betas=(0.9, 0.95))
    warmup = cfg["train"]["warmup_steps"]
    total_steps = max(1, len(train_loader) * cfg["train"]["epochs"])
    def lr_lambda(step):
        if step < warmup:
            return step / max(1, warmup)
        prog = (step - warmup) / max(1, total_steps - warmup)
        return 0.1 + 0.9 * 0.5 * (1 + math.cos(math.pi * prog))
    sched = torch.optim.lr_scheduler.LambdaLR(opt, lr_lambda)
    weights = LossWeights(**cfg["loss"])
    out_root = Path(cfg["out"]["root"])
    if rank == 0:
        (out_root / cfg["out"]["tb_dir"]).mkdir(parents=True, exist_ok=True)
        (out_root / cfg["out"]["ckpt_dir"]).mkdir(parents=True, exist_ok=True)
        tb = SummaryWriter(out_root / cfg["out"]["tb_dir"])
    step = 0
    best_val = float("inf")
    for epoch in range(cfg["train"]["epochs"]):
        model.train()
        for batch in train_loader:
            s = batch["s_in"].to(device, non_blocking=True)
            c = batch["c"].to(device, non_blocking=True)
            tgt = {k: v.to(device, non_blocking=True) for k, v in batch.items()
                   if isinstance(v, torch.Tensor) and k != "s_in" and k != "c"}
            with torch.autocast(device_type="cuda", dtype=torch.bfloat16):
                out = model(s, c)
                losses = compute_losses(out, tgt, weights)
            opt.zero_grad(set_to_none=True)
            losses["L_total"].backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), cfg["train"]["grad_clip"])
            opt.step()
            sched.step()
            if rank == 0 and step % cfg["train"]["log_every"] == 0:
                for k, v in losses.items():
                    tb.add_scalar(f"train/{k}", float(v.detach()), step)
                tb.add_scalar("lr", sched.get_last_lr()[0], step)
            step += 1
        # Val
        model.eval()
        agg: dict[str, float] = {}
        nb = 0
        with torch.no_grad():
            for batch in val_loader:
                s = batch["s_in"].to(device, non_blocking=True)
                c = batch["c"].to(device, non_blocking=True)
                tgt = {k: v.to(device, non_blocking=True) for k, v in batch.items()
                       if isinstance(v, torch.Tensor) and k != "s_in" and k != "c"}
                with torch.autocast(device_type="cuda", dtype=torch.bfloat16):
                    out = model(s, c)
                    losses = compute_losses(out, tgt, weights)
                for k, v in losses.items():
                    agg[k] = agg.get(k, 0.0) + float(v.detach())
                nb += 1
        if rank == 0:
            for k, v in agg.items():
                tb.add_scalar(f"val/{k}", v / max(1, nb), step)
            val_total = agg["L_total"] / max(1, nb)
            ckpt_path = Path(cfg["out"]["root"]) / cfg["out"]["ckpt_dir"]
            torch.save({"model": model.module.state_dict() if world > 1 else model.state_dict(),
                        "step": step, "epoch": epoch, "val_total": val_total,
                        "cfg": cfg},
                       ckpt_path / f"epoch_{epoch:03d}.pt")
            if val_total < best_val:
                best_val = val_total
                torch.save({"model": model.module.state_dict() if world > 1 else model.state_dict(),
                            "step": step, "epoch": epoch, "val_total": val_total,
                            "cfg": cfg},
                           ckpt_path / "best.pt")
    if world > 1:
        dist.destroy_process_group()

if __name__ == "__main__":
    main()
```

```python
import torch.nn.functional as F
def token_metrics(pred, tgt):
    mse = F.mse_loss(pred, tgt).item()
    cos = F.cosine_similarity(pred.flatten(-2), tgt.flatten(-2), dim=-1).mean().item()
    return {"token_mse": mse, "token_cos": cos}

# geom_metrics.py
def depth_l1(pred, tgt):
    return F.l1_loss(pred, tgt).item()
def chamfer_l1(pred, tgt):
    # both [B, k, 64, 64, 3] — pixel-aligned (decoded on same grid), so just L1
    return F.l1_loss(pred, tgt).item()
def pose_l2(pred, tgt):
    return F.mse_loss(pred, tgt).item() ** 0.5

# action_metrics.py
def action_metrics(pose_pred, grip_logit, action_tgt):
    pose_l2 = F.mse_loss(pose_pred, action_tgt[..., :6]).item() ** 0.5
    per_dim = ((pose_pred - action_tgt[..., :6]) ** 2).mean(dim=(0, 1)).sqrt().tolist()
    grip_pred = (grip_logit.sigmoid() > 0.5).float()
    grip_acc = (grip_pred == action_tgt[..., 6]).float().mean().item()
    return {"action_l2": pose_l2, "action_per_dim_l2": per_dim, "gripper_acc": grip_acc}
```

```python
from __future__ import annotations
import argparse, json
from pathlib import Path
import yaml, torch
from torch.utils.data import DataLoader, Subset
from wm3d.training.train import build_model, build_datasets
from wm3d.eval.token_metrics import token_metrics
from wm3d.eval.geom_metrics import depth_l1, chamfer_l1, pose_l2
from wm3d.eval.action_metrics import action_metrics

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--cfg", type=Path, required=True)
    ap.add_argument("--ckpt", type=Path, required=True)
    ap.add_argument("--out", type=Path,
                    default=Path("/home/user01/Minko/newwm/results/wm3d_v1/eval/offline.json"))
    args = ap.parse_args()
    cfg = yaml.safe_load(args.cfg.read_text())
    _, val_ds = build_datasets(cfg)
    loader = DataLoader(val_ds, batch_size=8, num_workers=4)
    model = build_model(cfg).cuda()
    sd = torch.load(args.ckpt, map_location="cuda")
    model.load_state_dict(sd["model"])
    model.eval()
    agg: dict[str, float] = {}
    nb = 0
    per_dim_acc = None
    with torch.no_grad():
        for batch in loader:
            s = batch["s_in"].cuda(non_blocking=True)
            c = batch["c"].cuda(non_blocking=True)
            tgt = {k: v.cuda(non_blocking=True) for k, v in batch.items()
                   if isinstance(v, torch.Tensor) and k not in {"s_in", "c"}}
            with torch.autocast(device_type="cuda", dtype=torch.bfloat16):
                out = model(s, c)
            m1 = token_metrics(out["pred_tokens"].float(), tgt["s_tgt"].float())
            m2 = {"depth_l1": depth_l1(out["depth"].float(), tgt["depth_tgt"].float()),
                  "chamfer_l1": chamfer_l1(out["point"].float(), tgt["point_tgt"].float()),
                  "pose_l2": pose_l2(out["pose_geom"].float(), tgt["pose_tgt"].float())}
            m3 = action_metrics(out["pose"].float(), out["gripper_logit"].float(), tgt["action_tgt"].float())
            for d in (m1, m2, m3):
                for k, v in d.items():
                    if isinstance(v, list):
                        if per_dim_acc is None:
                            per_dim_acc = [0.0] * len(v)
                        for i, val in enumerate(v):
                            per_dim_acc[i] += val
                    else:
                        agg[k] = agg.get(k, 0.0) + v
            nb += 1
    out_dict = {k: v / nb for k, v in agg.items()}
    if per_dim_acc:
        out_dict["action_per_dim_l2"] = [v / nb for v in per_dim_acc]
    args.out.parent.mkdir(parents=True, exist_ok=True)
    args.out.write_text(json.dumps(out_dict, indent=2))
    print(json.dumps(out_dict, indent=2))

if __name__ == "__main__":
    main()
```
